input: nCon(1);
var: stp(170), tgt(280);

{BIAS-1}
If T=0900 and dayofweek(date)=5 and (HighS(0) <> LowS(0))
	then buy ("LE-EoW") next bar at open;

If dayofweek(date)=0 and t>=2000 and marketposition = 1
	then sell ("LX-EoW") next bar at open;

{BIAS-2}
If (T>=100 and T<=300) and (dayofweek(date)=1 or dayofweek(date)=4) and (HighS(0) <> LowS(0))
	then sellshort ("SE-BIAS") next bar at lowest(l,2) stop;

If T=1000 and marketposition =-1 then buytocover ("SX-BIAS") next bar at open;

{USCITE}
If marketposition = 1 then begin
	if maxpositionprofit >= (stp*nCon) then sell ("LX-Breakeven") next bar at entryprice+(10*minmove/pricescale) stop;
end;

If marketposition =-1 then begin
	if maxpositionprofit >= (stp*nCon) then buytocover ("SX-Breakeven") next bar at entryprice-(10*minmove/pricescale) stop;
end;
